```php
<?php
// tests/Feature/Api/TaskApiTest.php

namespace Tests\Feature\Api;

use App\Models\Membership;
use App\Models\Project;
use App\Models\Task;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use PHPUnit\Framework\Attributes\Test;
use Tests\TestCase;

class TaskApiTest extends TestCase
{
    use RefreshDatabase;

    /**
     * タスクを作成できる
     */
    public function test_タスクを作成できる(): void
    {
        // ============================================
        // 1. Arrange（準備）
        // ============================================
        $user = User::factory()->create();
        $project = Project::factory()->create();

        // ユーザーをプロジェクトメンバーにする
        Membership::factory()->create([
            'user_id' => $user->id,
            'project_id' => $project->id,
            'role' => 'project_member',
        ]);

        // ============================================
        // 2. Act（実行）
        // ============================================
        $response = $this->actingAs($user)
            ->postJson("/api/projects/{$project->id}/tasks", [
                'title' => '新しいタスク',
                'description' => 'タスクの説明です',
            ]);

        // ============================================
        // 3. Assert（検証）
        // ============================================
        // レスポンスの確認
        $response->assertStatus(201);  // 201 Created
        $response->assertJson([
            'data' => [
                'title' => '新しいタスク',
                'description' => 'タスクの説明です',
                'status' => 'todo',
            ]
        ]);

        // データベースの確認
        $this->assertDatabaseHas('tasks', [
            'title' => '新しいタスク',
            'project_id' => $project->id,
            'created_by' => $user->id,
        ]);
    }
}
```

**👩‍💻ユーザー：** 「最後の `assertDatabaseHas` は何ですか？」

---

### 🤔 assertDatabaseHas の解説

**🐘ガネーシャ：** 「これは**データベースに実際に保存されてるか**を確認するメソッドや」

```
┌─────────────────────────────────────────────────────────────┐
│              assertDatabaseHas の意味                        │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  $this->assertDatabaseHas('tasks', [                        │
│      'title' => '新しいタスク',                             │
│      'project_id' => $project->id,                          │
│      'created_by' => $user->id,                             │
│  ]);                                                        │
│                                                             │
│  → tasks テーブルに、この条件に合うレコードがある？        │
│                                                             │
```

## 📖 第5章：テストをまとめて整理しよう

### 🎭 setUp メソッドで共通処理をまとめる

**🐘ガネーシャ：** 「ところでな、さっきのテスト見てみ。毎回同じような準備してへんか？」

**👩‍💻ユーザー：** 「確かに...User、Project、Membership を毎回作ってますね」

**🐘ガネーシャ：** 「こういう**共通の準備**は `setUp` メソッドにまとめられるんや」

```php
<?php
// tests/Feature/Api/TaskApiTest.php（リファクタリング後）

namespace Tests\Feature\Api;

use App\Models\Membership;
use App\Models\Project;
use App\Models\Task;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class TaskApiTest extends TestCase
{
    use RefreshDatabase;

    private User $user;
    private Project $project;

    /**
     * 各テストの前に実行される
     */
    protected function setUp(): void
    {
        parent::setUp();  // ← これは必ず書く！

        // 共通のテストデータを準備
        $this->user = User::factory()->create();
        $this->project = Project::factory()->create();

        Membership::factory()->create([
            'user_id' => $this->user->id,
            'project_id' => $this->project->id,
            'role' => 'project_member',
        ]);
    }

    /**
     * タスクを作成できる
     */
    public function test_タスクを作成できる(): void
    {
        // Arrange は setUp で完了してるので、すぐ Act へ！

        // Act
        $response = $this->actingAs($this->user)
            ->postJson("/api/projects/{$this->project->id}/tasks", [
                'title' => '新しいタスク',
                'description' => 'タスクの説明です',
            ]);

        // Assert
        $response->assertStatus(201);
        $response->assertJson([
            'data' => [
                'title' => '新しいタスク',
                'status' => 'todo',
            ]
        ]);
    }

    /**
     * todo ステータスのタスクを開始できる
     */
    public function test_todoステータスのタスクを開始できる(): void
    {
        // Arrange（タスクだけ追加で作成）
        $task = Task::factory()->create([
            'project_id' => $this->project->id,
            'created_by' => $this->user->id,
            'status' => 'todo',
        ]);

        // Act
        $response = $this->actingAs($this->user)
            ->postJson("/api/tasks/{$task->id}/start");

        // Assert
        $response->assertStatus(200);
        $response->assertJson([
            'data' => [
                'status' => 'doing',
            ]
        ]);
    }
}
```

```
┌─────────────────────────────────────────────────────────────┐
│              setUp メソッドの仕組み                          │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  【実行順序】                                               │
│                                                             │
│  test_タスクを作成できる が実行される時：                   │
│  1. setUp() が実行される（User, Project, Membership 作成）  │
│  2. test_タスクを作成できる() が実行される                  │
```
